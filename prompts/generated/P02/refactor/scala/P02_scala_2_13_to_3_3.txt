El siguiente codigo resuelve "Binary Search" en Scala version 2.13:

```scala
﻿// P02: Binary Search — Scala 2.13 (codigo base para refactoring)
// Estilo: bucle while con variables mutables (left, right), para forzar el desafio recursivo en el refactor
object P02BinarySearch_v213 {

  def binarySearch(arr: Array[Int], target: Int): Int = {
    var left = 0
    var right = arr.length - 1
    while (left <= right) {
      val mid = left + (right - left) / 2
      if (arr(mid) == target) return mid
      else if (arr(mid) < target) left = mid + 1
      else right = mid - 1
    }
    -1
  }

  def main(args: Array[String]): Unit = {
    assert(binarySearch(Array(1,3,5,7,9), 5) == 2)
    assert(binarySearch(Array(1,3,5,7,9), 4) == -1)
    assert(binarySearch(Array.empty[Int], 1) == -1)
    println("All tests passed")
  }
}
```
